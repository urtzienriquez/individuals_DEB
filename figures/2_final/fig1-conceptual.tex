% ============================================================
% Surface functions F(theta, T), theta,T in [0,1]
% ============================================================
% (1) Concave dome, fixed shape (curvature magnitude peaks at T=0.5, ->0 at edges)
\pgfmathdeclarefunction{Fconcave}{2}{%
  \pgfmathparse{2.5 * (1 - 4*(#1 - 0.5)^2) * (1 - 4*(#2 - 0.5)^2)}%
}
% (2) Gaussian, fixed shape (curvature magnitude peaks at T=0.5, decays at edges)
\pgfmathdeclarefunction{Fgauss}{2}{%
  \pgfmathparse{2.5 * exp(-12*(#1 - 0.5)^2 - 10*(#2 - 0.5)^2)}%
}
% (2b) Gaussian with T-opt shift: each theta peaks at a different T
\pgfmathdeclarefunction{FgaussTopt}{2}{%
  % Same max F=2.5 at every theta's T_opt: envelope + diagonal ridge +
  % compensation so the ridge crest is flat (all peaks equal height)
  \pgfmathparse{2.0 * exp(-15*(#2-0.5)^2 + 12*(#2-0.5)*(#1-0.5) - 1.8*(#1-0.5)^2)}%
}
% (3) Sigmoidal, FIXED inflection at theta=0.4 for all T (curvature magnitude only rescaled by T)
\pgfmathdeclarefunction{Fsig}{2}{%
  \pgfmathparse{(2.5/(1+exp(-10*(#1-0.4)))) * (1/(1+exp(-10*(#2-0.4))))}%
}
% (4) Sigmoidal with MOVING inflection c(T) = 0.2 + 0.6*T (curvature SIGN flips across T)
\pgfmathdeclarefunction{Fshift}{2}{%
  \pgfmathparse{2.5/(1+exp(-10*(#1-(0.2+0.6*#2))))}%
}
% Second-derivative-in-theta sign/shape functions (unnormalized, for row C)
% j1,j2,j3 >= 0 always; j4 changes sign at T=0.5
\pgfmathdeclarefunction{jConcave}{1}{\pgfmathparse{1 - 4*(#1-0.5)^2}}
\pgfmathdeclarefunction{jGauss}{1}{\pgfmathparse{exp(-10*(#1-0.5)^2)}}
\pgfmathdeclarefunction{jGaussTopt}{1}{\pgfmathparse{exp(-15*(#1-0.5)*(#1-0.5)) * (1 - 40*(#1-0.5)*(#1-0.5)) * (4*#1*(1-#1))}}
\pgfmathdeclarefunction{jSig}{1}{\pgfmathparse{(1+exp(-6))/(1+exp(-10*(#1-0.4)))}}
\pgfmathdeclarefunction{pShift}{1}{\pgfmathparse{1/(1+exp(-10*(0.5-(0.2+0.6*#1))))}}
\pgfmathdeclarefunction{jShift}{1}{\pgfmathparse{-4*(pShift(#1))*(1-pShift(#1))*(1-2*pShift(#1)) * (4*#1*(1-#1))}}
